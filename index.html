    <!-- Application Logic -->

    <script>

        const promptInput = document.getElementById('prompt');

        const motionStyle = document.getElementById('motionStyle');

        const durationVal = document.getElementById('durationVal');

        const generateBtn = document.getElementById('generateBtn');

        const previewContainer = document.getElementById('previewContainer');

        const placeholderText = document.getElementById('placeholderText');

        const loader = document.getElementById('loader');

        const canvas = document.getElementById('videoCanvas');

        const ctx = canvas.getContext('2d');

        const outputVideo = document.getElementById('outputVideo');

        const downloadBtn = document.getElementById('downloadBtn');


        // Update UI slider text

        durationInput.addEventListener('input', () => {

            durationVal.innerText = `${durationInput.value}s`;

        });


        generateBtn.addEventListener('click', async () => {

            if (!promptInput.value.trim()) {

                alert('Please enter a description or prompt for your video first!');

                return;

            }


            // UI State management

            generateBtn.disabled = true;

            placeholderText.classList.add('hidden');

            outputVideo.classList.add('hidden');

            downloadBtn.classList.add('hidden');

            loader.classList.remove('hidden');


            try {

                // Step 1: Fetch a free, dynamically generated base image via public Pollinations API

                const encodedPrompt = encodeURIComponent(promptInput.value.trim());

                const imageUrl = `https://pollinations.ai{encodedPrompt}?width=640&height=360&nologo=true&seed=${Math.floor(Math.random() * 100000)}`;

                

                const img = new Image();

                img.crossOrigin = "anonymous"; // Essential for saving canvas streams

                img.src = imageUrl;


                img.onload = () => {

                    loader.classList.add('hidden');

                    canvas.classList.remove('hidden');

                    animateAndRecord(img);

                };

            } catch (error) {

                alert('Error generating base assets. Try a slightly different prompt.');

                resetUI();

            }

        });


        function animateAndRecord(img) {

            const fps = 30;

            const duration = parseInt(durationInput.value);

            const totalFrames = fps * duration;

            let currentFrame = 0;

            const selectedMotion = motionStyle.value;


            // Setup MediaRecorder to record the HTML Canvas stream natively

            const stream = canvas.captureStream(fps);

            const mediaRecorder = new MediaRecorder(stream, { mimeType: 'video/webm;codecs=vp9' });

            const chunks = [];


            mediaRecorder.ondataavailable = (e) => {

                if (e.data.size > 0) chunks.push(e.data);

            };


            mediaRecorder.onstop = () => {

                const blob = new Blob(chunks, { type: 'video/mp4' });

                const videoURL = URL.createObjectURL(blob);

                

                // Show output player

                canvas.classList.add('hidden');

                outputVideo.src = videoURL;

                outputVideo.classList.remove('hidden');

                

                // Set download properties

                downloadBtn.href = videoURL;

                downloadBtn.download = `ai-video-${Date.now()}.mp4`;

                downloadBtn.classList.remove('hidden');


                generateBtn.disabled = false;

            };


            mediaRecorder.start();


            function draw() {

                if (currentFrame >= totalFrames) {

                    mediaRecorder.stop();

                    return;

                }


                ctx.clearRect(0, 0, canvas.width, canvas.height);

                const progress = currentFrame / totalFrames;


                let dx = 0, dy = 0, dWidth = canvas.width, dHeight = canvas.height;


                // Client-side visual motion mapping engines

                if (selectedMotion === 'pan-right') {

                    // Start slightly shifted, pan to center

                    const offset = 40 * (1 - progress);

                    ctx.drawImage(img, -offset, 0, canvas.width + 40, canvas.height);

                } else if (selectedMotion === 'zoom-in') {

                    // Incrementally increase dimensions from center frame anchor

                    const factor = 1 + (progress * 0.12); 

                    const w = canvas.width * factor;

                    const h = canvas.height * factor;

                    const x = (canvas.width - w) / 2;

                    const y = (canvas.height - h) / 2;

                    ctx.drawImage(img, x, y, w, h);

                } else if (selectedMotion === 'pulse') {

                    // Wave fluctuation scaling simulates cinematic slow breathing ambient light

                    const factor = 1 + (Math.sin(progress * Math.PI * 2) * 0.03);

                    const w = canvas.width * factor;

                    const h = canvas.height * factor;

                    const x = (canvas.width - w) / 2;

                    const y = (canvas.height - h) / 2;
